function tongHopLichHocThu() {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();

  var days = ["Thứ 2", "Thứ 3", "Thứ 4", "Thứ 5", "Thứ 6", "Thứ 7", "Chủ nhật"];
  var result = {};

  // Khởi tạo cấu trúc lưu trữ dữ liệu
  for (var d = 0; d < days.length; d++) {
    result[days[d]] = { "Quân trống": [], "Linh trống": [] };
  }

  // Hàm chuyển đổi chuỗi giờ thành số thập phân
  function parseTime(tStr) {
    var parts = tStr.toString().toLowerCase().split('h');
    var hours = parseInt(parts[0], 10) || 0;
    var mins = parseInt(parts[1], 10) || 0;
    return hours + (mins / 60);
  }

  // Hàm định dạng lại số thập phân thành chuỗi giờ
  function formatTime(decimalTime) {
    var h = Math.floor(decimalTime);
    var m = Math.round((decimalTime - h) * 60);
    return h + 'h' + (m > 0 ? m : '');
  }

  // Quét dữ liệu từ dòng 3 (index 2)
  for (var i = 2; i < data.length; i++) {
    var timeCell = data[i][1]; 
    if (!timeCell || timeCell.toString().indexOf('-') === -1) continue;

    var timeParts = timeCell.toString().split('-');
    var startT = parseTime(timeParts[0].trim());
    var endT = parseTime(timeParts[1].trim());

    for (var col = 2; col <= 8; col++) {
      var cellValue = data[i][col];
      if (cellValue) {
        var valStr = cellValue.toString().trim();
        if (valStr === "Quân trống" || valStr === "Linh trống") {
          result[days[col - 2]][valStr].push({start: startT, end: endT});
        }
      }
    }
  }

  // Thuật toán gộp các khoảng thời gian
  function mergeIntervals(intervals) {
    if (intervals.length === 0) return [];
    intervals.sort(function(a, b) { return a.start - b.start; });
    var merged = [intervals[0]];
    for (var i = 1; i < intervals.length; i++) {
      var current = intervals[i];
      var last = merged[merged.length - 1];
      if (current.start <= last.end) {
        last.end = Math.max(last.end, current.end);
      } else {
        merged.push(current);
      }
    }
    return merged;
  }

  var finalText = "";

  // Tổng hợp dữ liệu đầu ra
  for (var d = 0; d < days.length; d++) {
    var dayName = days[d];
    ["Quân trống", "Linh trống"].forEach(function(teacher) {
      var intervals = mergeIntervals(result[dayName][teacher]);
      if (intervals.length > 0) {
        var teacherName = teacher.replace(" trống", "");
        finalText += dayName + " (" + teacherName + "):\n";
        intervals.forEach(function(inv) {
          finalText += "Khung giờ từ " + formatTime(inv.start) + "-" + formatTime(inv.end) + "\n";
        });
        finalText += "\n";
      }
    });
  }

  // Đính kèm khối nội dung cố định
  finalText += "Lưu ý:\n";
  finalText += "- Học thử không mất phí\n";
  finalText += "- Thời gian học thử 30 phút\n";
  finalText += "- Được đánh giá giọng và lộ trình sau buổi học thử\n";

  // Xuất dữ liệu ra bảng thông báo trên giao diện Sheets (Hỗ trợ bôi đen và sao chép)
  var htmlContent = '<textarea style="width: 100%; height: 260px; font-family: sans-serif; font-size: 14px; padding: 5px; box-sizing: border-box;" readonly="readonly">' + finalText + '</textarea>';
  var htmlOutput = HtmlService.createHtmlOutput(htmlContent)
      .setWidth(450)
      .setHeight(320);
      
  SpreadsheetApp.getUi().showModalDialog(htmlOutput, 'Tổng Hợp Lịch Học Thử');
}
